    function clampCartToStock() {

      Object.keys(cart).forEach(id => {

        const item = products.find(p => p.id === id);

        const maxStock = item ? item.stock : 0;

        if (maxStock <= 0) {

          delete cart[id];

        } else if (cart[id] > maxStock) {

          cart[id] = maxStock;

        }

      });

    }

    function deductStockFromCart() {

      const overrides = getStockOverrides();

      Object.entries(cart).forEach(([id, qty]) => {

        if (qty <= 0) return;

        const item = products.find(p => p.id === id);

        if (!item) return;

        const nextStock = Math.max(0, item.stock - qty);

        item.stock = nextStock;

        overrides[item.id] = nextStock;

      });

      saveStockOverridesToStorage(overrides);

      cart = {};

    }

    function normalizeImageUrl(url) {

      if (!url) return PLACEHOLDER_IMG;

      const clean = url.trim();

      const driveMatch = clean.match(/drive\.google\.com\/(?:file\/d\/|open\?id=|uc\?id=)([a-zA-Z0-9_-]+)/);

      if (driveMatch && driveMatch[1]) {

        return `https://lh3.googleusercontent.com/d/${driveMatch[1]}=w800`;

      }

      return clean;

    }

    function parseCsv(text) {

      const rows = [];

      let row = [];

      let field = "";

      let inQuotes = false;

      for (let i = 0; i < text.length; i++) {

        const c = text[i];

        const next = text[i + 1];

        if (inQuotes) {

          if (c === '"' && next === '"') {

            field += '"';

            i++;

          } else if (c === '"') {

            inQuotes = false;

          } else {

            field += c;

          }

        } else {

          if (c === '"') {

            inQuotes = true;

          } else if (c === ',') {

            row.push(field.trim());

            field = "";

          } else if (c === '\n' || (c === '\r' && next === '\n')) {

            if (c === '\r') i++;

            row.push(field.trim());

            if (row.some(cell => cell !== "")) rows.push(row);

            row = [];

            field = "";

          } else {

            field += c;

          }

        }

      }

      if (field !== "" || row.length > 0) {

        row.push(field.trim());

        if (row.some(cell => cell !== "")) rows.push(row);

      }

      return rows;

    }

    async function loadProductsFromSheet() {

      const sheetUrl = (CONFIG.store.googleSheetCsvUrl || "").trim();

      const syncEl = document.getElementById("sync-status");

      if (!sheetUrl) {

        syncEl.textContent = "Baki stok dipotong automatik selepas pesanan dihantar ke WhatsApp.";

        return;

      }

      try {

        const cacheBuster = sheetUrl.includes("?") ? `&t=${Date.now()}` : `?t=${Date.now()}`;

        const response = await fetch(sheetUrl + cacheBuster);

        if (!response.ok) throw new Error("Gagal memuat turun data Google Sheets");

        const csvText = await response.text();

        const rows = parseCsv(csvText);

        if (rows.length < 2) throw new Error("Tiada rekod produk dalam Google Sheets");

        const headers = rows[0].map(h => h.toLowerCase().trim());

        const idx = (key) => headers.indexOf(key.toLowerCase());

        const parsed = rows.slice(1).map((r, i) => {

          const name = r[idx("name")] || "";

          if (!name) return null;

          const statusVal = (r[idx("status")] || "aktif").toLowerCase();

          const rawStock = idx("stock") >= 0 ? parseInt((r[idx("stock")] || "0").replace(/[^0-9-]/g, ""), 10) : 99;

          const isDisabled = statusVal === "habis" || statusVal === "tidak aktif" || statusVal === "off";

          const finalStock = isDisabled ? 0 : Math.max(0, isNaN(rawStock) ? 0 : rawStock);

          return {

            id: r[idx("id")] || `sheet-${i + 1}`,

            name: name,

            category: r[idx("category")] || "Lain-lain",

            description: r[idx("description")] || "",

            retailPrice: parseFloat((r[idx("retailprice")] || "0").replace(/[^0-9.]/g, "")) || 0,

            agentPrice: parseFloat((r[idx("agentprice")] || "0").replace(/[^0-9.]/g, "")) || 0,

            stock: finalStock,

            image: normalizeImageUrl(r[idx("image")])

          };

        }).filter(Boolean);

        if (parsed.length > 0) {

          products = parsed;

          applyStockOverrides();

          syncEl.innerHTML = `<span class="text-emerald-700 font-medium"><i class="fas fa-sync-alt mr-1"></i>Inventori diselaraskan dengan Google Sheets (${products.length} item)</span>`;

          renderCategoryTabs();

          renderCatalog();

          updateSummaryUI();

        }

      } catch (err) {

        console.warn("Menggunakan data sandaran:", err);

        syncEl.textContent = "Baki stok dipotong automatik selepas pesanan dihantar ke WhatsApp.";

      }

    }

    function changeQty(id, delta) {

      const item = products.find(p => p.id === id);

      if (!item) return;

      const current = cart[id] || 0;

      const next = current + delta;

      if (delta > 0 && next > item.stock) {

        alert(`Maaf, baki stok untuk "${item.name}" hanya tinggal ${item.stock} item.`);

        return;

      }

      if (next <= 0) {

        delete cart[id];

      } else {

        cart[id] = next;

      }

      renderCatalog();

      updateSummaryUI();

    }

      modal.classList.add("hidden");

      modal.classList.remove("flex");

    }

    function setFulfillment(method) {

      fulfillmentMethod = method;

      const btnPickup = document.getElementById("btn-pickup");

      const btnDelivery = document.getElementById("btn-delivery");

      const pickupBox = document.getElementById("pickup-box");

      const deliveryBox = document.getElementById("delivery-box");

      if (method === "pickup") {

        btnPickup.className = "border-2 border-emerald-600 bg-emerald-50 text-emerald-900 rounded-2xl p-3 text-left transition";

        btnDelivery.className = "border-2 border-slate-200 bg-white text-slate-700 rounded-2xl p-3 text-left transition";

        pickupBox.classList.remove("hidden");

        deliveryBox.classList.add("hidden");

      } else {

        btnDelivery.className = "border-2 border-emerald-600 bg-emerald-50 text-emerald-900 rounded-2xl p-3 text-left transition";

        btnPickup.className = "border-2 border-slate-200 bg-white text-slate-700 rounded-2xl p-3 text-left transition";

        deliveryBox.classList.remove("hidden");

        pickupBox.classList.add("hidden");

      }

    }

    function validatePickupDate() {

      const dateInput = document.getElementById("pickup-date");

      const errEl = document.getElementById("pickup-error");

      if (!dateInput.value) {

        errEl.classList.add("hidden");

        return true;

      }

      const [y, m, d] = dateInput.value.split("-").map(Number);

      const dayIndex = new Date(y, m - 1, d).getDay();

      if (CONFIG.store.closedDays.includes(dayIndex)) {

        errEl.textContent = `Maaf, kedai tutup pada hari ${DAY_NAMES[dayIndex]}. Sila pilih hari lain.`;

        errEl.classList.remove("hidden");

        dateInput.value = "";

        return false;

      }

      errEl.classList.add("hidden");

      return true;

    }

    function submitWhatsAppOrder() {

      const { totalQty, isAgent, totalAmount } = getOrderSummary();

      if (totalQty === 0) {

        alert("Sila pilih sekurang-kurangnya satu item aiskrim.");

        return;

      }

      // Pastikan kuantiti troli tidak melebihi baki stok semasa

      for (const [id, qty] of Object.entries(cart)) {

        if (qty <= 0) continue;

        const item = products.find(p => p.id === id);

        if (!item || qty > item.stock) {

          alert(`Stok untuk "${item ? item.name : id}" tidak mencukupi. Baki semasa: ${item ? item.stock : 0}.`);

          clampCartToStock();

          renderCatalog();

          updateSummaryUI();

          return;

        }

      }

      const name = document.getElementById("cust-name").value.trim();

      if (!name) {

        alert("Sila masukkan nama anda.");

        document.getElementById("cust-name").focus();

        return;

      }

      let fulfillmentText = "";

      if (fulfillmentMethod === "pickup") {

        const isValidDay = validatePickupDate();

        const dateVal = document.getElementById("pickup-date").value;

        const slotVal = document.getElementById("pickup-slot").value;

        if (!isValidDay || !dateVal || !slotVal) {

          alert("Sila pilih tarikh buka kedai dan slot masa ambil sendiri.");

          return;

        }

        const [y, m, d] = dateVal.split("-");

        fulfillmentText = `*Kaedah:* Ambil Sendiri (Pickup)\n*Tarikh:* ${d}/${m}/${y}\n*Slot Masa:* ${slotVal}`;

      } else {

        const address = document.getElementById("cust-address").value.trim();

        const note = document.getElementById("cust-note").value.trim();

        if (!address) {

          alert("Sila masukkan alamat penghantaran.");

          document.getElementById("cust-address").focus();

          return;

        }

        fulfillmentText = `*Kaedah:* Penghantaran (Delivery)\n*Alamat:* ${address}`;

        if (note) fulfillmentText += `\n*Catatan:* ${note}`;

      }

      const lines = Object.entries(cart)

        .filter(([, q]) => q > 0)

        .map(([id, qty], idx) => {

          const item = products.find(p => p.id === id);

          const unit = isAgent ? item.agentPrice : item.retailPrice;

          return `${idx + 1}. ${item.name}\n   ${qty} × ${formatMoney(unit)} = *${formatMoney(unit * qty)}*`;

        })

        .join("\n");

      const message = [

        `Assalamualaikum / Hai *${CONFIG.store.title}*, saya ingin membuat pesanan:`,

        "",

        `*Nama:* ${name}`,

        fulfillmentText,

        "",

        "*Senarai Pesanan:*",

        lines,

        "",

        `*Jumlah Item:* ${totalQty} item`,

        `*Kategori Harga:* ${isAgent ? "Harga Ejen (>=5 item)" : "Harga Runcit"}`,

        `*Jumlah Bayaran:* *${formatMoney(totalAmount)}*`

      ].join("\n");

      const url = `https://wa.me/${CONFIG.store.whatsappNumber}?text=${encodeURIComponent(message)}`;

      // Potong baki stok automatik selepas pesanan disahkan dihantar ke WhatsApp

      deductStockFromCart();

      renderCatalog();

      updateSummaryUI();

      closeCart();

      window.open(url, "_blank", "noopener");

    }

    initStoreContent();

    applyStockOverrides();

    renderCategoryTabs();

    renderCatalog();

    updateSummaryUI();

    loadProductsFromSheet();

  </script>
